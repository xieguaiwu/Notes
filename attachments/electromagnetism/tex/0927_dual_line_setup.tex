  \foreach \x in {-3.4,-2.3,-1.2,1.3,2.4,3.5}

  \node[teal] at (-1.2,2.25) {$\vec{E}$};

  % ---------- 轨迹抛物线 y = -x^2/8 ----------

  % 初速度
  \draw[->,very thick,blue] (0,0) -- (1.2,0) node[above=1pt] {$\vec{v}_0$};
  % 电场力（向下）
  \draw[->,very thick,red] (0,0) -- (0,-1.2);
  \node[below=3pt] at (0,-1.25) {$\vec{F}=-q\vec{E}_{\mathrm{net}}$};
